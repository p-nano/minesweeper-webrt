// 必要: npm install ws const WebSocket = require("ws"); const wss = new WebSocket.Server({ port: 8081 }); let clients = new Map(); // id -> ws let nextId = 1; wss.on("connection", (ws) => { const myID = nextId++; clients.set(myID, ws); // 接続したクライアントに自分のIDと既存クライアントを通知 ws.send(JSON.stringify({ type: "init", id: myID, connected: Array.from(clients.keys()) })); // 他のクライアントにも通知（新規接続） for (let [id, client] of clients) { if (id !== myID) { client.send(JSON.stringify({ type: "new-peer", id: myID })); } } ws.on("message", (data) => { let message; try { message = JSON.parse(data); } catch (e) { console.error("Invalid message:", data); return; } // 送信先が指定されている場合だけ転送 if (message.to && clients.has(message.to)) { clients.get(message.to).send(JSON.stringify(message)); } }); ws.on("close", () => { clients.delete(myID); // 他のクライアントに切断を通知 for (let client of clients.values()) { client.send(JSON.stringify({ type: "peer-left", id: myID })); } }); }); console.log("Signaling server running on ws://localhost:8081");
